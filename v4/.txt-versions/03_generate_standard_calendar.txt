/*==============================================================================
  STANDARD CALENDAR GENERATOR

  This script creates a comprehensive standard Gregorian calendar view that enriches
  the date spine with a wide range of date attributes and calculations. It serves
  as the foundation for all other calendar components in the system.

  The resulting calendar view includes:
  - Date keys and descriptive attributes
  - Year, quarter, month, week, and day attributes
  - First/last day calculations
  - Day of week/month/year counts
  - Weekend flags
  - Same day/date last year references
  - Time components for sub-day grains

  USAGE:
  ------
  CALL GENERATE_CALENDAR('DAY');

  PARAMETERS:
  -----------
  TIME_GRAIN: The granularity level (SECOND, MINUTE, HOUR, DAY, MONTH, YEAR)

  RETURNS:
  --------
  A string indicating success or failure with details

  PREREQUISITES:
  -------------
  Requires the DATE_SPINE_{TIME_GRAIN} table to exist

  AUTHOR: Andrew Exley, Lument Pty Ltd
  DATE CREATED: 2025-04-04
  LAST MODIFIED: 2025-04-05 (Corrected structure)
==============================================================================*/

CREATE OR REPLACE PROCEDURE GENERATE_CALENDAR(TIME_GRAIN VARCHAR(10))
RETURNS STRING
LANGUAGE JAVASCRIPT
AS
$$
    // ----- Input Validation -----
    const validTimeGrains = ['SECOND', 'MINUTE', 'HOUR', 'DAY', 'MONTH', 'YEAR'];
    const grainUpper = TIME_GRAIN.toUpperCase();
    if (validTimeGrains.indexOf(grainUpper) === -1) {
        return "ERROR: Invalid time grain. Must be one of: SECOND, MINUTE, HOUR, DAY, MONTH, YEAR";
    }

    // ----- Dynamic Name Construction -----
    const baseTableName = "DATE_SPINE_" + grainUpper; // Name of the source date spine table
    const viewName = "CALENDAR_" + grainUpper;       // Name of the calendar view to be created

    // Declare SQL parts outside the try block
    let createViewSql = '';
    let columnDefinitions = ''; // For column names and comments: col_name COMMENT '...'
    let selectExpressions = ''; // For SELECT list: expression AS col_name

    try {
        // ----- Column Definition and SELECT Expression Logic -----

        // --- Initialise Lists ---
        let colDefs = []; // Array to hold "COLUMN_NAME COMMENT '...'" strings
        let selExprs = []; // Array to hold "expression AS COLUMN_NAME" strings

        // --- Helper function to add column ---
        function addColumn(name, expression, comment) {
            // Escape single quotes in comments for SQL safety
            const escapedComment = comment.replace(/'/g, "''");
            colDefs.push(`${name} COMMENT '${escapedComment}'`);
            selExprs.push(`${expression} AS ${name}`);
        }

        // --- Define ALL Columns using addColumn ---
        // Core date columns - available for all grains
        addColumn('DATE_TIME', 'ds.DATE_TIME', 'The base timestamp at the specified grain (TIMESTAMP_NTZ).');
        addColumn('DATE_KEY', 'DATE(ds.DATE_TIME)', 'The date part of the timestamp (DATE, YYYY-MM-DD). Primary key candidate.');
        addColumn('FULL_DATE_DESCRIPTION', "TO_VARCHAR(DATE(ds.DATE_TIME), 'DD MMMM YYYY')", 'Human-readable full date (e.g., "27 October 2023").');
        addColumn('YEAR', 'YEAR(ds.DATE_TIME)', 'The four-digit year (e.g., 2023).');

        // Year-Based attributes
        addColumn('IS_LEAP_YEAR', '(MOD(YEAR(ds.DATE_TIME), 4) = 0 AND MOD(YEAR(ds.DATE_TIME), 100) != 0 OR MOD(YEAR(ds.DATE_TIME), 400) = 0)', 'Boolean flag indicating if the year is a leap year.');
        addColumn('FIRST_DAY_OF_YEAR', "DATE(DATE_TRUNC('YEAR', ds.DATE_TIME))", 'The first day of the year (DATE).');
        addColumn('LAST_DAY_OF_YEAR', "DATE(DATEADD(day, -1, DATEADD(year, 1, DATE_TRUNC('YEAR', ds.DATE_TIME))))", 'The last day of the year (DATE).');

        // Quarter-Based (Not applicable for YEAR grain)
        if (grainUpper !== 'YEAR') {
            addColumn('QUARTER', 'QUARTER(ds.DATE_TIME)', 'Quarter number of the year (1-4).');
            addColumn('QUARTER_NAME', "'Q' || QUARTER(ds.DATE_TIME)::STRING", 'Descriptive quarter name (e.g., "Q1").');
            addColumn('FIRST_DAY_OF_QUARTER', "DATE(DATE_TRUNC('QUARTER', ds.DATE_TIME))", 'The first day of the quarter (DATE).');
            addColumn('LAST_DAY_OF_QUARTER', "DATE(LAST_DAY(ds.DATE_TIME, 'QUARTER'))", 'The last day of the quarter (DATE).');
        }

        // Month-Based (Not applicable for YEAR grain)
        if (grainUpper !== 'YEAR') {
            addColumn('MONTH', 'MONTH(ds.DATE_TIME)', 'Month number (1-12).');
            addColumn('MONTH_NAME', "TO_VARCHAR(ds.DATE_TIME, 'MMMM')", 'Full name of the month (e.g., "January").');
            addColumn('MONTH_NAME_SHORT', "TO_VARCHAR(ds.DATE_TIME, 'Mon')", 'Abbreviated month name (e.g., "Jan").');
            addColumn('MONTH_SORT', 'MONTH(ds.DATE_TIME)', 'Month number for sorting MonthName (1-12).');
            addColumn('FIRST_DAY_OF_MONTH', "DATE(DATE_TRUNC('MONTH', ds.DATE_TIME))", 'The first day of the month (DATE).');
            addColumn('LAST_DAY_OF_MONTH', "DATE(LAST_DAY(ds.DATE_TIME, 'MONTH'))", 'The last day of the month (DATE).');
            addColumn('MONTH_YEAR', "TO_VARCHAR(ds.DATE_TIME, 'YYYY-MM')", 'Year and month combination for grouping (e.g., "2023-10").');
        }

        // Week/Day-Based (Vary by grain)
        if (['DAY', 'HOUR', 'MINUTE', 'SECOND'].includes(grainUpper)) {
             addColumn('DAY_OF_YEAR', 'DAYOFYEAR(ds.DATE_TIME)', 'Sequential day number within the year (1-366).');
             addColumn('DAY_OF_MONTH', 'DAY(ds.DATE_TIME)', 'Day number within the month (1-31).');
             addColumn('DAY_OF_WEEK', 'DAYOFWEEKISO(ds.DATE_TIME)', 'ISO day number within the week (Monday=1, Sunday=7).');
             addColumn('DAY_OF_WEEK_NAME', "TO_VARCHAR(ds.DATE_TIME, 'Day')", 'Full name of the day of the week (e.g., "Monday").');
             addColumn('DAY_OF_WEEK_NAME_SHORT', "TO_VARCHAR(ds.DATE_TIME, 'Dy')", 'Abbreviated day name (e.g., "Mon").');
             addColumn('DAY_OF_WEEK_SORT', 'DAYOFWEEKISO(ds.DATE_TIME)', 'ISO day number for sorting DayOfWeekName (1-7).');
             addColumn('WEEK_OF_YEAR', 'WEEKOFYEAR(ds.DATE_TIME)', 'Week number within the year (using Snowflake default, often ISO-like).');
             addColumn('IS_WEEKEND', '(DAYOFWEEKISO(ds.DATE_TIME) IN (6, 7))', 'Boolean flag indicating if the day is a Saturday or Sunday.');
             addColumn('FIRST_DAY_OF_WEEK', "DATE(DATE_TRUNC('WEEK', ds.DATE_TIME))", 'The first day (Monday) of the ISO week (DATE).');
             addColumn('LAST_DAY_OF_WEEK', "DATE(DATEADD(day, 6, DATE_TRUNC('WEEK', ds.DATE_TIME)))", 'The last day (Sunday) of the ISO week (DATE).');
             addColumn('SAME_DATE_LAST_YEAR', "DATE(DATEADD(year, -1, DATE(ds.DATE_TIME)))", 'The exact calendar date one year prior (handles leap years).');
             addColumn('SAME_DAY_LAST_YEAR',
                       "DATE(DATEADD(day, DAYOFWEEKISO(DATE(ds.DATE_TIME)) - 1, DATE_TRUNC('WEEK', DATE(DATEADD(year, -1, DATE(ds.DATE_TIME))))))",
                       'The date falling on the same ISO day of the week in the same ISO week number relative to the date one year prior.');
        } else if (grainUpper === 'MONTH') {
             addColumn('DAY_OF_YEAR', 'DAYOFYEAR(ds.DATE_TIME)', 'Sequential day number within the year (always 1st of month).');
             addColumn('DAY_OF_MONTH', 'DAY(ds.DATE_TIME)', 'Day number within the month (always 1).');
             addColumn('SAME_DATE_LAST_YEAR', "DATE(DATEADD(year, -1, DATE(ds.DATE_TIME)))", 'The exact calendar date one year prior.');
             // SAME_DAY_LAST_YEAR is less clearly defined for MONTH grain start date, omitting for clarity.
        } else if (grainUpper === 'YEAR') {
             addColumn('DAY_OF_YEAR', 'DAYOFYEAR(ds.DATE_TIME)', 'Sequential day number within the year (always 1).');
             addColumn('SAME_DATE_LAST_YEAR', "DATE(DATEADD(year, -1, DATE(ds.DATE_TIME)))", 'The exact calendar date one year prior.');
             // SAME_DAY_LAST_YEAR is not meaningful at YEAR grain level.
        }

        // Time Component Columns (Applicable for finer grains)
        if (['HOUR', 'MINUTE', 'SECOND'].includes(grainUpper)) {
             addColumn('TIME', 'TIME(ds.DATE_TIME)', 'The time part of the timestamp.');
        }
        switch (grainUpper) {
            case 'SECOND':
                addColumn('HOUR', 'HOUR(ds.DATE_TIME)', 'Hour of the day (0-23).');
                addColumn('MINUTE', 'MINUTE(ds.DATE_TIME)', 'Minute of the hour (0-59).');
                addColumn('SECOND', 'SECOND(ds.DATE_TIME)', 'Second of the minute (0-59).');
                break;
            case 'MINUTE':
                addColumn('HOUR', 'HOUR(ds.DATE_TIME)', 'Hour of the day (0-23).');
                addColumn('MINUTE', 'MINUTE(ds.DATE_TIME)', 'Minute of the hour (0-59).');
                break;
            case 'HOUR':
                addColumn('HOUR', 'HOUR(ds.DATE_TIME)', 'Hour of the day (0-23).');
                break;
        }

        // ----- Join Definitions and Expressions -----
        columnDefinitions = colDefs.join(',\n    ');
        selectExpressions = selExprs.join(',\n    ');

        // ----- View Creation -----
        const viewComment = `Standard Gregorian calendar view based on ${grainUpper} grain date spine. Contains comprehensive date and time attributes for business reporting and analysis.`;
        // *** NOW define createViewSql using the populated variables ***
        createViewSql = `
            CREATE OR REPLACE VIEW ${viewName} (
                ${columnDefinitions}
            )
            COMMENT = '${viewComment.replace(/'/g, "''")}'
            AS
            SELECT
                ${selectExpressions}
            FROM ${baseTableName} ds -- Added alias 'ds' for clarity
            ORDER BY ds.DATE_TIME
        `;

        // Execute the CREATE VIEW statement
        let createViewStmt = snowflake.createStatement({
            sqlText: createViewSql
        });
        createViewStmt.execute();

        return `Standard Gregorian calendar view '${viewName}' generated successfully based on '${baseTableName}'. Access the view with SELECT * FROM ${viewName}.`;

    } catch (err) {
        return `ERROR creating standard calendar view '${viewName}': ${err.code} - ${err.message}\nAttempted SQL:\n${createViewSql}`;
    }
$$;
